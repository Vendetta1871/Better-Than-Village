// A villager sees a zombie villager and calls the healer, who cures it at once with a potion of weakness and a golden apple
locate structure #minecraft:village
forceload add {LX-96} {LZ-96} {LX+96} {LZ+96}
#wait-log Registered generated vanilla village 60
#sleep 40
tick sprint 2400
#wait-log Sprint completed 400
execute positioned {LX} 100 {LZ} run btv status
// The cure in the treasury (the healer's workshop makes it from spider eyes, sugar, mushrooms, gold and apples)
execute positioned {LX} 100 {LZ} run btv world stock minecraft:golden_apple 1
execute positioned {LX} 100 {LZ} run btv world stock minecraft:splash_potion 1 minecraft:weakness
time set 1000
execute positioned {LX-8} 100 {LZ-10} positioned over motion_blocking run summon minecraft:zombie_villager ~ ~ ~ {NoAI:1b,PersistenceRequired:1b,ArmorItems:[{},{},{},{id:"minecraft:leather_helmet",count:1}]}
#sleep 2
#wait-log calls the healer 60
#wait-log a healer of .* cures a zombie villager 120
#wait-log a zombie villager is cured 30
execute if entity @e[type=minecraft:zombie_villager]
execute if entity @e[type=minecraft:villager]
// At night the call wakes the healer
execute positioned {LX} 100 {LZ} run btv world stock minecraft:golden_apple 1
execute positioned {LX} 100 {LZ} run btv world stock minecraft:splash_potion 1 minecraft:weakness
time set 18000
#sleep 20

#wait-log calls the healer 60
#wait-log a healer of .* cures a zombie villager 120
